Args: res (resource), width (int), alt, class, eager */ -}} {{- $res := .res }} {{- if $res }} {{- $w := .width | default 800 }} {{- $small := $res.Resize (printf "%dx webp" (cond (lt $res.Width $w) $res.Width $w)) }} {{- $large := $res.Resize (printf "%dx webp" (cond (lt $res.Width (mul $w 2)) $res.Width (mul $w 2))) }} {{- end }}
